Using @oasisomniverse/web6-api
This demo loads the npm package via esm.sh. The same code runs in Node.js or any bundler.
Install
npm install @oasisomniverse/web6-api
AI Completion with BRAID
import { Web6Client } from '@oasisomniverse/web6-api'; const web6 = new Web6Client({ baseUrl: 'https://api.web6.oasisomniverse.one' }); web6.setToken('<your-oasis-avatar-key>'); const { result } = await web6.completion.complete({ Provider: 'auto', Model: 'auto', Messages: [{ Role: 'user', Content: 'Your prompt here' }], UseHolonicBraid: true, // Generator/Solver split — 74× cost reduction UseFAHRN: true, FahrnTaskType: 'general', Routing: { Priority: 'cost', Fallback: true }, }); console.log(result.Content); console.log(`$${result.CostUsd} via ${result.Provider}/${result.Model}`);
FAHRN Multi-Agent Dispatch
const { result } = await web6.fahrn.solve({ Problem: 'Design a zero-downtime Postgres migration for a 50M-row table', TaskType: 'architecture', Mode: 'Parallel', // Serial | Parallel | Debate | Voting | Decomposed MaxTotalTokens: 8000, MaxCostUsd: 0.50, }); console.log(result.Answer); console.log('Agents used:', result.AgentsUsed); console.log('BRAID graph reused:', result.BraidGraphReused);
OpenServ Provider — one SERV key, all models
// Seed the agent registry from OpenServ's live SERV catalogue await fetch('https://api.web6.oasisomniverse.one/v1/reasoning-network/agents/seed-openserv', { method: 'POST', headers: { Authorization: 'Bearer <key>', 'Content-Type': 'application/json' }, body: '{}' } ); // Route a completion through OpenServ const { result } = await web6.completion.complete({ Provider: 'openserv', // GPT-5, Claude, Gemini, Grok, DeepSeek via SERV gateway Model: 'auto', Messages: [{ Role: 'user', Content: 'Hello via OpenServ' }], });
Streaming SSE
const stream = await web6.completion.stream({ Provider: 'auto', Messages: [{ Role: 'user', Content: 'Write a long technical document...' }], }); for await (const chunk of stream) { process.stdout.write(chunk.delta); }
